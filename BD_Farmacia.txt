-- #area de creacion de tablas -------------------------------------------------------------------------------------------
-- #Creacion de la tabla usuario++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++
CREATE TABLE usuario(
    id INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    password VARCHAR(100) NOT NULL,
    tipo_documento VARCHAR(20) NOT NULL,
    documento VARCHAR(20) NOT NULL UNIQUE,
    direccion VARCHAR(150),
    telefono VARCHAR(20),
    estado BOOLEAN NOT NULL DEFAULT TRUE,
    admin BOOLEAN NOT NULL DEFAULT FALSE
) AUTO_INCREMENT = 10000;

SELECT * FROM usuario;

-- #Creacion de la tabla Categoria+++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++

CREATE TABLE categoria (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL,
    descripcion VARCHAR(200)
) AUTO_INCREMENT = 10000;

-- #Creacion de la tabla producto+++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++
CREATE TABLE producto (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    prin_act VARCHAR(100),
    marca VARCHAR(100),
    lab VARCHAR(100),
    presentacion VARCHAR(100),
    precio DECIMAL(10,2) NOT NULL,
    requiere_receta BOOLEAN NOT NULL,
    activo BOOLEAN NOT NULL,
    
    id_categoria INT NOT NULL,
    
    FOREIGN KEY (id_categoria)
        REFERENCES categoria(id)
) AUTO_INCREMENT = 10000;

-- #Creacion de la tabla compra+++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++

CREATE TABLE compra (
    cod INT PRIMARY KEY AUTO_INCREMENT,
    fecha DATE NOT NULL,
    estado BOOLEAN NOT NULL DEFAULT TRUE,
    
    id_usuario INT NOT NULL,
    
    FOREIGN KEY (id_usuario)
        REFERENCES usuario(id)
) AUTO_INCREMENT = 10000;

-- #Creacion de la tabla de venta--------------------------------------------------------------------------------------------------

CREATE TABLE venta (
    cod INT AUTO_INCREMENT PRIMARY KEY,
    fecha DATE NOT NULL,
    
    id_usuario INT NOT NULL,
    
    FOREIGN KEY (id_usuario)
        REFERENCES usuario(id)
) AUTO_INCREMENT = 10000;

-- #Creacion de la tabla lote------------------------------------------------------------------------------------------------------

CREATE TABLE lote (
    id INT AUTO_INCREMENT PRIMARY KEY,
    numero_lote VARCHAR(50) NOT NULL,
    fecha_vencimiento DATE NOT NULL,
    stock_actual INT NOT NULL,
    
    id_producto INT NOT NULL,
    
    FOREIGN KEY (id_producto)
        REFERENCES producto(id)
) AUTO_INCREMENT = 10000;

-- #Creacion de la tabla detalle de compra-------------------------------------------------------------------------------------------

CREATE TABLE detalle_compra (
    cod INT AUTO_INCREMENT PRIMARY KEY,
    cant INT NOT NULL,
    costo_uni DECIMAL(10,2) NOT NULL,
    
    id_compra INT NOT NULL,
    id_producto INT NOT NULL,
    id_lote INT NOT NULL,
    
    FOREIGN KEY (id_compra)
        REFERENCES compra(cod),
        
    FOREIGN KEY (id_producto)
        REFERENCES producto(id),
        
    FOREIGN KEY (id_lote)
        REFERENCES lote(id)
) AUTO_INCREMENT = 10000;

-- #Creacion de la tabla detalle de venta--------------------------------------------------------------------------------------------------

CREATE TABLE detalle_venta (
    cod INT AUTO_INCREMENT PRIMARY KEY,
    cant INT NOT NULL,
    precio_unitario DECIMAL(10,2) NOT NULL,
    
    id_venta INT NOT NULL,
    id_producto INT NOT NULL,
    id_lote INT NOT NULL,
    
    FOREIGN KEY (id_venta)
        REFERENCES venta(cod),
        
    FOREIGN KEY (id_producto)
        REFERENCES producto(id),
        
    FOREIGN KEY (id_lote)
        REFERENCES lote(id)
) AUTO_INCREMENT = 10000;

-- creacion del procedure buscar producto por nombre --------------------------------------------------------------------------------
CREATE PROCEDURE SP_BUSCAR_PRODUCTO_NOMBRE(
    p_nombre VARCHAR(100)
)
SELECT
    p.id,
    p.nombre,
    p.prin_act,
    p.marca,
    p.lab,
    p.presentacion,
    p.precio,
    p.requiere_receta,
    p.activo,
    c.id AS id_categoria,
    c.nombre AS nom_categoria
FROM producto p
INNER JOIN categoria c
ON p.id_categoria = c.id
WHERE p.nombre LIKE CONCAT('%', p_nombre, '%')
ORDER BY p.nombre;

-- creacion del procedure actualizar precio producto -------------------------------------------------------------------
DELIMITER //

CREATE PROCEDURE SP_ACTUALIZAR_PRECIO_PRODUCTO_20(
    IN p_id_producto INT
)
BEGIN
    UPDATE producto
    SET precio = (
        SELECT ROUND(IFNULL(MAX(dc.costo_uni), 0) * 1.20, 2)
        FROM detalle_compra dc
        WHERE dc.id_producto = p_id_producto
    )
    WHERE id = p_id_producto;
END //

-- listar lotes disponibles de un producto ordenados por vencimiento -----------------------------------------------------------
DELIMITER //

CREATE PROCEDURE SP_LISTAR_LOTES_PRODUCTO(
    IN p_id_producto INT
)
BEGIN
    SELECT
        l.id,
        l.numero_lote,
        l.fecha_vencimiento,
        l.stock_actual,
        p.id AS id_producto,
        p.nombre
    FROM lote l
    INNER JOIN producto p
        ON l.id_producto = p.id
    WHERE l.id_producto = p_id_producto
      AND l.stock_actual > 0
    ORDER BY l.fecha_vencimiento ASC;
END //

DELIMITER ;

-- creacion del procedur listar compra detalladas ------------------------------------------------------------------------
DELIMITER //
CREATE PROCEDURE SP_LISTAR_DETALLE_COMPRA()
BEGIN
    SELECT 
        c.cod AS id_compra,
        c.fecha,
        u.nombre AS nombre_usuario,
        p.id AS id_producto,
        p.nombre AS nombre_producto,
        dc.cant,
        dc.costo_uni,
        l.numero_lote,
        dc.cod AS id_detalle
    FROM compra c
    INNER JOIN usuario u ON c.id_usuario = u.id
    INNER JOIN detalle_compra dc ON c.cod = dc.id_compra
    INNER JOIN producto p ON dc.id_producto = p.id
    INNER JOIN lote l ON dc.id_lote = l.id
    ORDER BY c.cod DESC;
END //
DELIMITER ;

DELIMITER //

    IN p_idDetalle INT
)

    SELECT
        dc.cod AS id_detalle,
        dc.cant,
        dc.costo_uni,

        c.fecha,
        c.estado,

        u.id AS id_usuario,
        u.nombre AS nombre_usuario,

        p.id AS id_producto,

        p.prin_act,

        p.precio,
        p.requiere_receta,

        cat.id AS id_categoria,

        cat.descripcion,

        l.id AS id_lote,

        l.stock_actual

    FROM detalle_compra dc
    INNER JOIN compra c
        ON dc.id_compra = c.cod
    INNER JOIN usuario u
        ON c.id_usuario = u.id
    INNER JOIN producto p
        ON dc.id_producto = p.id

        ON p.id_categoria = cat.id

-- creacion porcedure detalle compra inventario ------------------------------------------------------------------------
DELIMITER //

CREATE PROCEDURE SP_LISTAR_DETALLE_COMPRA_INVENTARIO()
BEGIN
    SELECT
        l.id AS id_lote,
        l.numero_lote,
        p.nombre AS nombre_producto,
        l.stock_actual,
        l.fecha_vencimiento,
        dc.cant AS stock_inicial,
        dc.costo_uni,
        (dc.cant * dc.costo_uni) AS costo_lote,
        c.fecha AS fecha_adquisicion,
        u.nombre AS nombre_usuario,
        p.id AS id_producto,
        dc.cod AS id_detalle,
        c.cod AS id_compra,
        cat.id AS id_categoria,       -- NUEVO
        cat.nombre AS nombre_categoria -- NUEVO
    FROM detalle_compra dc
    INNER JOIN compra c ON dc.id_compra = c.cod
    INNER JOIN usuario u ON c.id_usuario = u.id
    INNER JOIN producto p ON dc.id_producto = p.id
    INNER JOIN categoria cat ON p.id_categoria = cat.id -- NUEVO JOIN
    INNER JOIN lote l ON dc.id_lote = l.id
    ORDER BY c.fecha DESC, l.fecha_vencimiento ASC, l.numero_lote ASC;
END //

-- #GUI Movimientos -----------------------------------------------------------------------------------------------------
-- creacion del procedure compras recientes -----------------------------------------------------------------------------
DELIMITER //

CREATE PROCEDURE SP_MOVIMIENTOS_COMPRAS()
BEGIN
    SELECT
        c.cod AS id_compra,
        c.fecha,

        (
            SELECT IFNULL(SUM(dc2.cant * dc2.costo_uni), 0)
            FROM detalle_compra dc2
            WHERE dc2.id_compra = c.cod
        ) AS total_compra,

        dc.cod AS id_detalle,
        dc.cant,
        dc.costo_uni,
        (dc.cant * dc.costo_uni) AS subtotal,

        p.id AS id_producto,
        p.nombre AS nombre_producto,
        p.precio AS precio_actual,
        p.marca,
        p.lab,

        cat.nombre AS nombre_categoria,

        l.id AS id_lote,
        l.numero_lote,
        l.fecha_vencimiento,
        l.stock_actual,

        u.id AS id_usuario,
        u.nombre AS nombre_usuario

    FROM compra c
    INNER JOIN detalle_compra dc
        ON c.cod = dc.id_compra
    INNER JOIN producto p
        ON dc.id_producto = p.id
    INNER JOIN categoria cat
        ON p.id_categoria = cat.id
    INNER JOIN lote l
        ON dc.id_lote = l.id
    INNER JOIN usuario u
        ON c.id_usuario = u.id
    ORDER BY c.cod DESC, dc.cod ASC;
END //

DELIMITER ;

-- creacion del procedure ventas recientes ------------------------------------------------------------------------------
DELIMITER //

CREATE PROCEDURE SP_MOVIMIENTOS_VENTAS()
BEGIN
    SELECT
        v.cod AS id_venta,
        v.fecha,

        (
            SELECT IFNULL(SUM(dv2.cant * dv2.precio_unitario), 0)
            FROM detalle_venta dv2
            WHERE dv2.id_venta = v.cod
        ) AS total_venta,

        dv.cod AS id_detalle,
        dv.cant,
        dv.precio_unitario,
        (dv.cant * dv.precio_unitario) AS subtotal,

        p.id AS id_producto,
        p.nombre AS nombre_producto,
        p.precio AS precio_actual,
        p.marca,
        p.lab,

        cat.nombre AS nombre_categoria,

        l.id AS id_lote,
        l.numero_lote,
        l.fecha_vencimiento,
        l.stock_actual,

        u.id AS id_usuario,
        u.nombre AS nombre_usuario

    FROM venta v
    INNER JOIN detalle_venta dv
        ON v.cod = dv.id_venta
    INNER JOIN producto p
        ON dv.id_producto = p.id
    INNER JOIN categoria cat
        ON p.id_categoria = cat.id
    INNER JOIN lote l
        ON dv.id_lote = l.id
    INNER JOIN usuario u
        ON v.id_usuario = u.id
    ORDER BY v.cod DESC, dv.cod ASC;
END //

DELIMITER ;
